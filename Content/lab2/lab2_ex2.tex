\textbf{Report 2: Mã nguồn trong hàm ngắt Timer Callback}

Đoạn mã xử lý quét 4 LED 7 đoạn và đảo trạng thái 2 LED \texttt{DOT} triển khai trong hàm \texttt{HAL\_TIM\_PeriodElapsedCallback}:

\begin{lstlisting}[language=C, caption={Hàm ngắt Timer điều khiển quét 4 LED 7 đoạn và chớp tắt 2 LED DOT}, label={lst:lab2_ex2_code}]
/* USER CODE BEGIN 4 */
int led = 0;
int counter = 50;
int counter1 = 100;

void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
    if (htim->Instance == TIM2)
    {
        counter--;
        counter1--;

        if (counter1 <= 0)
        {
        	counter1 = 100;
        	HAL_GPIO_TogglePin ( DOT_GPIO_Port , DOT_Pin );
        }

        if (counter <= 0)
        {
            counter = 50;

            if (led == 0)
            {
                HAL_GPIO_WritePin(GPIOA, EN1_Pin, GPIO_PIN_SET);
                HAL_GPIO_WritePin(GPIOA, EN2_Pin, GPIO_PIN_SET);
                HAL_GPIO_WritePin(GPIOA, EN3_Pin, GPIO_PIN_SET);
                display7SEG(1);
                HAL_GPIO_WritePin(GPIOA, EN0_Pin, GPIO_PIN_RESET);

                led = 1;
            }
            else if (led == 1)
            {
            	HAL_GPIO_WritePin(GPIOA, EN0_Pin, GPIO_PIN_SET);
            	HAL_GPIO_WritePin(GPIOA, EN2_Pin, GPIO_PIN_SET);
            	HAL_GPIO_WritePin(GPIOA, EN3_Pin, GPIO_PIN_SET);
                display7SEG(2);
                HAL_GPIO_WritePin(GPIOA, EN1_Pin, GPIO_PIN_RESET);

                led = 2;
            }
            else if (led == 2)
            {
            	HAL_GPIO_WritePin(GPIOA, EN0_Pin, GPIO_PIN_SET);
            	HAL_GPIO_WritePin(GPIOA, EN1_Pin, GPIO_PIN_SET);
            	HAL_GPIO_WritePin(GPIOA, EN3_Pin, GPIO_PIN_SET);
                display7SEG(3);
                HAL_GPIO_WritePin(GPIOA, EN2_Pin, GPIO_PIN_RESET);

                led = 3;
            }
            else
            {
            	HAL_GPIO_WritePin(GPIOA, EN0_Pin, GPIO_PIN_SET);
            	HAL_GPIO_WritePin(GPIOA, EN1_Pin, GPIO_PIN_SET);
            	HAL_GPIO_WritePin(GPIOA, EN2_Pin, GPIO_PIN_SET);
                display7SEG(0);
                HAL_GPIO_WritePin(GPIOA, EN3_Pin, GPIO_PIN_RESET);

                led = 0;
            }
        }
    }
}
/* USER CODE END 4 */

\end{lstlisting}

\textbf{Short question: Tần số của quá trình quét LED là bao nhiêu?}
\begin{itemize}
    \item \textbf{Trả lời:} 
    Chu kỳ quét cho từng LED 7 đoạn là $T_S = 0.5\text{s}$ ($500\text{ms}$). Do hệ thống quét luân phiên toàn bộ $4$ LED 7 đoạn, tổng chu kỳ để quét xong 1 lượt tất cả các LED là:
    $$T_{scan} = 4 \times T_S = 4 \times 0.5\text{s} = 2.0\text{s}$$
    Tần số của quá trình quét (Scanning Frequency) là:
    $$f_{scan} = \frac{1}{T_{scan}} = \frac{1}{2.0\text{s}} = 0.5\text{Hz}$$
\end{itemize}
