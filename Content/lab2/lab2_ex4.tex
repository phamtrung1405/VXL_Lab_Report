\textbf{Report 1: Mã nguồn trong hàm ngắt Timer Callback}

Mã nguồn điều chỉnh biến đếm \texttt{counter} về $25$ ($25 \times 10\text{ms} = 250\text{ms}$) để đạt tần số quét $1\text{Hz}$ triển khai trong hàm \texttt{HAL\_TIM\_PeriodElapsedCallback}:

\begin{lstlisting}[language=C, caption={Cấu hình quét 4 LED 7 đoạn tần số 1Hz trong hàm ngắt Timer}, label={lst:lab2_ex4_callback}]
int counter = 25;
int counter1 = 100;

void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
    if (htim->Instance == TIM2)
    {
        counter--;
        counter1--;

        if (counter1 <= 0)
        {
        	counter1 = 100;
        	HAL_GPIO_TogglePin ( DOT_GPIO_Port , DOT_Pin );
        }

        if (counter <= 0)
        {
            counter = 25;
            update7SEG(index_led ++);
            if (index_led > MAX_LED - 1) index_led = 0;
        }
    }
}
\end{lstlisting}
