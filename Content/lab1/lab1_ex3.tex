\subsection{Exercise 3 - Lab 1}

\textbf{Mục tiêu \& Yêu cầu:}
\begin{itemize}
    \item Mở rộng hệ thống thành ngã tư 4 hướng (4-way traffic light).
    \item Sắp xếp $12$ đèn LED (gồm 4 đèn Đỏ, 4 đèn Vàng, 4 đèn Xanh) tạo thành giao lộ 4 hướng trực quan.
    \item Lập trình điều khiển đồng bộ giữa 2 trục đường (Trục Đông - Tây và Trục Bắc - Nam):
    \begin{itemize}
        \item Khi trục Đông - Tây là đèn \texttt{RED} ($5$s), trục Bắc - Nam sẽ chạy \texttt{GREEN} ($3$s) $\rightarrow$ \texttt{YELLOW} ($2$s).
        \item Khi trục Bắc - Nam chuyển sang đèn \texttt{RED} ($5$s), trục Đông - Tây sẽ chạy \texttt{GREEN} ($3$s) $\rightarrow$ \texttt{YELLOW} ($2$s).
    \end{itemize}
\end{itemize}

\textbf{Report 2: Đoạn chương trình thực thi}

Mã nguồn triển khai trong vòng lặp \texttt{while(1)} điều khiển luân phiên 2 trục đường ngã tư:

\begin{lstlisting}[language=C, caption={Chương trình điều khiển hệ thống đèn giao thông ngã tư 4 hướng}, label={lst:lab1_ex3_code}]
/* USER CODE BEGIN WHILE */
while (1)
{
    HAL_GPIO_WritePin ( LED_GREEN_ROW_GPIO_Port , LED_GREEN_ROW_Pin , GPIO_PIN_RESET );
    HAL_GPIO_WritePin ( LED_YELLOW_ROW_GPIO_Port , LED_YELLOW_ROW_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_RED_ROW_GPIO_Port , LED_RED_ROW_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_GREEN_COL_GPIO_Port , LED_GREEN_COL_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_YELLOW_COL_GPIO_Port , LED_YELLOW_COL_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_RED_COL_GPIO_Port , LED_RED_COL_Pin , GPIO_PIN_RESET );
    HAL_Delay (3000) ;

    HAL_GPIO_WritePin ( LED_GREEN_ROW_GPIO_Port , LED_GREEN_ROW_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_YELLOW_ROW_GPIO_Port , LED_YELLOW_ROW_Pin , GPIO_PIN_RESET );
    HAL_GPIO_WritePin ( LED_RED_ROW_GPIO_Port , LED_RED_ROW_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_GREEN_COL_GPIO_Port , LED_GREEN_COL_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_YELLOW_COL_GPIO_Port , LED_YELLOW_COL_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_RED_COL_GPIO_Port , LED_RED_COL_Pin , GPIO_PIN_RESET );
    HAL_Delay (2000) ;

    HAL_GPIO_WritePin ( LED_GREEN_ROW_GPIO_Port , LED_GREEN_ROW_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_YELLOW_ROW_GPIO_Port , LED_YELLOW_ROW_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_RED_ROW_GPIO_Port , LED_RED_ROW_Pin , GPIO_PIN_RESET );
    HAL_GPIO_WritePin ( LED_GREEN_COL_GPIO_Port , LED_GREEN_COL_Pin , GPIO_PIN_RESET );
    HAL_GPIO_WritePin ( LED_YELLOW_COL_GPIO_Port , LED_YELLOW_COL_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_RED_COL_GPIO_Port , LED_RED_COL_Pin , GPIO_PIN_SET );
    HAL_Delay (3000) ;

    HAL_GPIO_WritePin ( LED_GREEN_ROW_GPIO_Port , LED_GREEN_ROW_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_YELLOW_ROW_GPIO_Port , LED_YELLOW_ROW_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_RED_ROW_GPIO_Port , LED_RED_ROW_Pin , GPIO_PIN_RESET );
    HAL_GPIO_WritePin ( LED_GREEN_COL_GPIO_Port , LED_GREEN_COL_Pin , GPIO_PIN_SET );
    HAL_GPIO_WritePin ( LED_YELLOW_COL_GPIO_Port , LED_YELLOW_COL_Pin , GPIO_PIN_RESET );
    HAL_GPIO_WritePin ( LED_RED_COL_GPIO_Port , LED_RED_COL_Pin , GPIO_PIN_SET );
    HAL_Delay (2000) ;
/* USER CODE END WHILE */

}
